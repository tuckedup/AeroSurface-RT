cmake_minimum_required(VERSION 3.20)
project(aerosurface_perception)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
set(ONNXRUNTIME_ROOT "" CACHE PATH "ONNX Runtime SDK root")
find_path(ORT_INCLUDE onnxruntime_cxx_api.h PATHS ${ONNXRUNTIME_ROOT}/include REQUIRED)
find_library(ORT_LIBRARY onnxruntime PATHS ${ONNXRUNTIME_ROOT}/lib REQUIRED)
set(CORE_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../../../cpp)
add_executable(perception_node src/perception_node.cpp ${CORE_ROOT}/src/runtime.cpp)
target_include_directories(perception_node PRIVATE ${CORE_ROOT}/include ${ORT_INCLUDE})
target_link_libraries(perception_node ${ORT_LIBRARY})
ament_target_dependencies(perception_node rclcpp sensor_msgs diagnostic_msgs)
target_compile_options(perception_node PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS perception_node DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/image_publisher.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_package()
